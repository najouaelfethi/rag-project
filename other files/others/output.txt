Self-attention, sometimes called intra-attention is an attention mechanism relating different positions of a single sequence in order to compute a representation of the sequence. Self-attention has been used successfully in a variety of tasks including reading comprehension, abstractive summarization, textual entailment and learning task-independent sentence representations [4, 22, 23, 19]. #determine if text might be meaningful content to analyse
def should_analyze(text):
    #skip short text
    if len(text.split())<50:
        return False
    #skip text that's mostly special characters
    if sum(1 for char in text if char in ['@', '[', ']', '#', '*']) > len(text) * 0.3:
        return False
    
    return True


def analyze_with_llm(text):
    classifier = pipeline("zero-shot-classification",model="facebook/bart-large-mnli")
    #possible categories for he text
    candidate_labels = [
        "main content",
        "header",
        "footer",
        "reference",
        "other"
    ]
    result=classifier(text[:512], candidate_labels, hypothesis_template="This text is about {}.")
    #get most likely label
    content_type = result['labels'][0]
    core_content = content_type == "main content" and len(text.split()) > 100
    return {"content_type":content_type,"core_content":core_content}


def enrich_metadata(chunk):
    #Basic metadata without llm
    metadata={
        "word_count": len(chunk.page_content.split()),
        "has_special_chars": any(char in chunk.page_content for char in ['@', '[', ']', '#', '*']),
        "is_short":len(chunk.page_content.split())<50
    }

    #using LLM for potentially menaingful chunks
    if should_analyze(chunk.page_content):
        llm_analysis=analyze_with_llm(chunk.page_content)
        metadata.update(llm_analysis)
    else:
        metadata.update({
            "content_type":"unknown",
            "core_content":False
        })

    return metadata
